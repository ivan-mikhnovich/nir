% --- 1a. Что изменилось после первой защиты ---
\begin{frame}{Что изменилось после первой защиты}
  \begin{itemize}
    \item \structure{Единое правило отбора пар для всех подходов}: раньше эмбеддинговые
      подходы давали по одному соответствию на \emph{исходный} узел, GNN-матчер~---
      на \emph{целевой}, а LLM~--- жадно-инъективное отображение 1:1, и колонки
      сравнительной таблицы были несопоставимы. Теперь все подходы проходят через
      одно правило, а варианты правила вынесены в отдельную абляцию

    \vspace{0.3em}
    \item \structure{Честная агрегация LLM}: раньше строка LLM была максимумом по трём
      представлениям, выбранным по эталону оцениваемой пары. Теперь публикуются числа
      без выбора по паре: среднее по режимам, выбор режима по схеме leave-one-pair-out
      и ансамбль по правилу «не менее двух из трёх»

    \vspace{0.3em}
    \item \structure{Проверяемый негативный результат по GNN}: конфигурация и сиды
      сохраняются, матчер оценки исправлен, добавлен протокол с полностью отложенной
      онтологией

    \vspace{0.3em}
    \item \structure{Полный перепрогон}: все подходы пересчитаны на тех же 21 паре,
      числа сведены в один набор артефактов; проверки согласованности, метрики и
      правила отбора покрыты юнит-тестами в \texttt{tests/}
  \end{itemize}
\end{frame}

% --- 2. Постановка задачи ---
\begin{frame}{Постановка задачи}
  \begin{columns}[T,onlytextwidth]
    \column{0.45\textwidth}
      \structure{\large Таксономии}
      \vspace{0.3em}

      \begin{itemize}
        \item Т.н. <<дерево классификатора>>
        \item Узлы~--- классы (например, \texttt{Document}, \texttt{Paper})
        \item Рёбра~--- отношение «is-a» (\texttt{subClassOf})
        \item В общем случае DAG, а не дерево
        \item Можно рассматривать как частный случай онтологии
      \end{itemize}

    \column{0.52\textwidth}
      \structure{\large Сопоставление таксономий}
      \begin{itemize}
        \item Даны две таксономии \texttt{A} и \texttt{B}
        \item Нужно найти соответствия между их классами
        \item Один и тот же концепт~--- разные имена, разная иерархия
        \item Результат: alignment = список матчей (1:1)
      \end{itemize}

      \vspace{0.3em}
      \structure{\large Сложность}
      \begin{itemize}
        \item Синонимы: \texttt{Person} $\leftrightarrow$ \texttt{Human}
        \item Разная гранулярность: \texttt{Paper} $\leftrightarrow$ \texttt{FullPaper}?
        \item Отсутствие общей лексики в текстовых описаниях узлов ~--- нужно понимать смысл написанного
      \end{itemize}
  \end{columns}

  \vspace{0.3em}
  \footnotesize
  \structure{Вторая половина темы:} проверить согласованность~--- как внутри каждой
  таксономии, так и в найденном отображении.
\end{frame}

% --- 6. Метрики ---
\begin{frame}{Метрики оценки}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \structure{\large Основная метрика: F1}
      \vspace{0.3em}

      \begin{itemize}
        \item 1:1 matching (жадно-инъективное правило отбора)
        \item Precision = TP / (TP + FP)
        \item Recall = TP / (TP + FN)
        \item F1 = гармоническое среднее
      \end{itemize}

      \vspace{0.5em}
      \structure{\large Дополнительные метрики}
      \begin{itemize}
        \item Precision и Recall отдельно
        \item True Positives (количество правильных пар)
        \item Predicted count vs. Ground Truth count
      \end{itemize}

    \column{0.48\textwidth}
      \structure{\large Baseline: StringEquiv}
      \vspace{0.3em}

      \begin{itemize}
        \item Точное совпадение имён классов без учёта регистра букв
        \item Официальный lower bound OAEI
        \item Если имена совпадают~--- матч
        \item На некоторых парах онтологий достигает F1 = 1{,}0
        \item Ломается на синонимах и любых различиях в лексике
      \end{itemize}

      \vspace{0.5em}
      \begin{center}
        \color{amber}\large
        Любой разумный подход к этой задаче должен быть не хуже, чем StringEquiv
      \end{center}
  \end{columns}
\end{frame}

% --- 7. Подходы (обзор) ---
\begin{frame}{Подходы}
  \begin{enumerate}
    \item \structure{Embedding-based}

      {\small BERT (LaBSE, MiniLM, ruRoberta, rubert) + cosine similarity.\\
      Результаты сильно зависят от того, какой текст мы извлекаем из узлов.\\
      Оказалось что \texttt{name\_only} на OAEI работает лучше, чем имя + путь.}

    \vspace{0.5em}
    \item \structure{LLM-based (LLMs4OM)}

      {\small DeepSeek-v4-flash / GPT-4.1-mini.  Бинарная классификация
      пар-кандидатов при помощи LLM (yes/no).  Три представления: C, CP, CC.}

    \vspace{0.5em}
    \item \structure{GraphSAGE (GNN)}

      {\small Siamese GraphSAGE на MiniLM-фичах, leave-one-pair-out CV.\\
      Без порога: GNN 0{,}333 против MiniLM 0{,}349;
      с калибровкой: 0{,}667 против 0{,}653; парного теста нет.}

    \vspace{0.5em}
    \item \structure{Ablation study: правило отбора и ретриверы}

      {\small Правило отбора пар влияет на F1 сильнее выбора модели: на одних и тех же
      эмбеддингах MiniLM \texttt{argmax} даёт 0{,}309, жадно-инъективное 1:1~--- 0{,}349,
      \texttt{dedupe}~--- 0{,}497, а оно же с порогом 0{,}5--0{,}7~--- 0{,}559--0{,}660.
      Ретриверы под честной агрегацией: BM25~--- 0{,}624, MiniLM~--- 0{,}602,
      гибрид~--- \textbf{0{,}645} (лучший LLM-результат, но ниже строкового baseline 0{,}663).}
  \end{enumerate}
\end{frame}

% --- 8. Embedding-based подход ---
\begin{frame}{Embedding-based: BERT + cosine similarity}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \structure{\large Схема}
      \begin{enumerate}
        \item Имя класса $\to$ sentence-transformer $\to$ эмбеддинг
        \item Cosine similarity каждого source-узла со всеми target-узлами
        \item 1:1 greedy matching по убыванию сходства
        \item Четыре модели: LaBSE, MiniLM, ruRoberta-large, rubert-base
      \end{enumerate}

      \vspace{0.2em}
      \structure{\large Интересное наблюдение}
      \begin{itemize}
        \item Полное описание узла (путь, атрибуты, несовместимости) \alert{ухудшает} F1:
          0{,}309 у одного имени против 0{,}211 в полном описании (MiniLM)
      \end{itemize}

    \column{0.48\textwidth}
      \structure{\large Результаты под единым правилом 1:1 (21 пара)}
      \vspace{0.2em}

      {\scriptsize
      \begin{tabular}{@{}lccc@{}}
        \toprule
        Модель & Dim & Без порога & С порогом \\
        \midrule
        LaBSE & 768 & 0{,}360 & 0{,}636 \\
        MiniLM & 384 & 0{,}349 & 0{,}653 \\
        ruRoberta-large & 1024 & 0{,}299 & \textbf{0{,}677} \\
        rubert-base & 768 & 0{,}285 & 0{,}646 \\
        \midrule
        StringEquiv & --- & 0{,}663 & --- \\
        \bottomrule
      \end{tabular}}

      \vspace{0.2em}
      \structure{\large Порог калибруется под модель}
      \begin{itemize}
        \item Схема leave-one-pair-out: 0{,}70--0{,}90 у LaBSE, 0{,}80--0{,}90
          у MiniLM, 0{,}95 у rubert-base и ruRoberta-large
        \item Единый порог не переносится: у LaBSE 680 из 986 пар слабее 0{,}7,
          у ruRoberta-large сходства лежат в 0{,}73--1{,}0
        \item Лучший средний F1 работы~--- ruRoberta-large с порогом 0{,}677 против
          0{,}663 у StringEquiv
      \end{itemize}
  \end{columns}
\end{frame}

% --- 9. LLM-based подход ---
\begin{frame}{LLM-based: адаптация LLMs4OM}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \structure{\large Адаптация LLMs4OM (arXiv 2404.10317)}
      \begin{enumerate}
        \item MiniLM: полное описание; эмбеддинговая часть гибрида: имя.
          Один запрос на узел со всеми кандидатами
        \item C: имя, атрибуты, комментарий, несовместимости;
          CP добавляет цепочку предков, CC~--- детей
        \item Самооценка LLM $>0{,}7$ (не вероятность), затем отбор 1:1
        \item Ансамбль~--- наше расширение; шаг статьи $S_{ir}>0{,}9$ не реализован
      \end{enumerate}

      \vspace{0.5em}
      \structure{\large Модели}
      \begin{itemize}
        \item \textbf{GPT-4.1-mini} (OpenAI-совместимый API)
        \item \textbf{DeepSeek-v4-flash} (DeepSeek Platform)
      \end{itemize}

    \column{0.48\textwidth}
      \structure{\large Агрегация режимов: средний F1, 21 пара}
      \vspace{0.2em}

      {\scriptsize
      \begin{tabular}{@{}lcccc@{}}
        \toprule
        Конфигурация & Среднее & LOO & Ансамбль & \textit{max} \\
        \midrule
        BM25 + DeepSeek & 0{,}604 & 0{,}651 & 0{,}624 & \textit{0{,}685} \\
        MiniLM + DeepSeek & 0{,}559 & 0{,}593 & 0{,}602 & \textit{0{,}636} \\
        MiniLM + GPT & 0{,}546 & 0{,}585 & 0{,}575 & \textit{0{,}604} \\
        Гибрид + DeepSeek & 0{,}620 & 0{,}644 & \textbf{0{,}645} & \textit{0{,}708} \\
        \bottomrule
      \end{tabular}}

      \vspace{0.2em}
      \begin{itemize}
        \item Публикуются только правила без выбора по эталону пары: среднее по режимам,
          выбор режима по leave-one-pair-out, ансамбль «не менее двух из трёх»
        \item Курсивный столбец \textit{max}~--- верхняя граница (режим выбирается по
          эталону пары), как результат не заявляется
        \item Лучший ансамбль~--- гибрид 0{,}645, это \alert{ниже} baseline 0{,}663
      \end{itemize}

      \vspace{0.1em}
      {\scriptsize \textbf{Стоимость:} 3648 обращений на конфигурацию (1365~--- у BM25).}
  \end{columns}
\end{frame}

\begin{frame}{Ablation study: ретриверы при честной агрегации}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \structure{\large Что проверили}
      \begin{itemize}
        \item \textbf{BM25}: лексический ретривер + DeepSeek
        \item \textbf{MiniLM}: эмбеддинговый ретривер + DeepSeek / GPT
        \item \textbf{Гибрид}: union top-K (MiniLM + BM25) + DeepSeek
      \end{itemize}

      \vspace{0.4em}
      \structure{\large Результат (ансамбль $\geq$2 из 3)}
      \begin{itemize}
        \item Гибрид~--- 0{,}645, BM25~--- 0{,}624, MiniLM+DS~--- 0{,}602,
          MiniLM+GPT~--- 0{,}575
        \item Тесты \emph{среднего по режимам}, не ансамблей:
          гибрид минус BM25 $+0{,}017$, BM25 минус MiniLM $+0{,}045$;
          оба $p_{\rm Holm}=1{,}0$ (семейство из 12 сравнений)
      \end{itemize}

    \column{0.46\textwidth}
      \structure{\large Интерпретация}
      \begin{itemize}
        \item По числу пар единственный лучший ансамбль: гибрид 7/21, BM25 6/21,
          DeepSeek(MiniLM) 3/21, GPT 3/21, ещё две пары~--- ничья
        \item Recall кандидатов зависит от ретривера; вклад лексики и структуры
          отдельно не изолирован
        \item Размах повторных прогонов одной пары достигает 0{,}217;
          это не оценка неопределённости разности макро-F1
      \end{itemize}

      \vspace{0.2em}
      \begin{center}
        \color{amber}\large
        Агрегация и ретривер~--- разные факторы сравнения
      \end{center}
  \end{columns}
\end{frame}

\begin{frame}{GraphSAGE: как обучали и что получилось}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \structure{\large Идея}
      \begin{itemize}
        \item MiniLM знает \textbf{имя} класса (\texttt{Paper})
        \item GNN добавляет знание о \textbf{соседях} (\texttt{Document $>$ Paper})
        \item Гипотеза: структура графа помогает сопоставлению
      \end{itemize}

      \vspace{0.3em}
      \structure{\large Как обучали}
      \begin{itemize}
        \item Правильная пара (\texttt{Paper}$\leftrightarrow$\texttt{Paper}):
          цель~--- косинус $\to$ 1{,}0
        \item Случайная пара (\texttt{Paper}$\leftrightarrow$\texttt{Reviewer}):
          цель~--- косинус $\to$ 0{,}0
        \item Три случайных негатива на каждый позитив
        \item 21 фолд leave-one-pair-out: 20 пар учим, одну проверяем
      \end{itemize}

    \column{0.44\textwidth}
      \structure{\large Итог (средний F1 по 21 паре)}

      {\scriptsize
      \begin{tabular}{@{}lcc@{}}
        \toprule
        Подход & Без порога & С порогом \\
        \midrule
        MiniLM & 0{,}349 & 0{,}653 \\
        GraphSAGE (pair-LOO) & 0{,}333 & 0{,}667 \\
        \midrule
        \texttt{--linear} & 0{,}314 & --- \\
        \texttt{--no-synthetic} & 0{,}345 & --- \\
        \texttt{ontology-loo} & 0{,}318 & --- \\
        \bottomrule
      \end{tabular}}

      \vspace{0.3em}
      \structure{\large Границы интерпретации}
      \begin{itemize}
        \item Без порога GNN ниже MiniLM, с калибровкой~--- выше;
          парного теста этой разницы нет.
        \item Малый объём данных и различия структуры~--- возможные объяснения,
          а не установленные причины.
        \item Pair-LOO сохраняет знакомые онтологии; ontology-LOO
          отдельно проверяет полностью отложенную онтологию.
      \end{itemize}
  \end{columns}
\end{frame}

% --- 9a. Проверка согласованности ---
\begin{frame}{Частичная структурная проверка}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \structure{\large Что проверяется}
      \begin{itemize}
        \item \textbf{Внутри таксономии}: циклы «is-a», совпадающие метки, несовместимость класса с собственным предком, обрыв ссылки на родителя, отсутствие пути к корню
        \item \textbf{В отображении}: инъективность 1:1, противоречие объявленной несовместимости, несохранение подчинения

% --- 10a. Ограничения ---
\begin{frame}{Ограничения}
  \begin{itemize}
    \item \structure{Проверки согласованности неполны}: формального вывода нет, полнота ограничена
      несовместимостью, объявленной в самих онтологиях

    \vspace{0.3em}
    \item \structure{Эмбеддинговые модели не дообучались}: использованы предобученные,
      резерв дообучения под задачу не исчерпан

    \vspace{0.3em}
    \item \structure{Измерен только recall ретривера при разных $k$}:
      основные LLM-прогоны~--- гибрид $k=10$, остальные $k=5$;
      чувствительность классификатора к $k$ не измерена

    \vspace{0.3em}
    \item \structure{Разброс генерации LLM} измерен на двух парах:
      размах трёх прогонов достигает $0{,}217$; полного набора повторов
      и доказательства снижения дисперсии ансамблем нет

    \vspace{0.3em}
    \item \structure{Pair-LOO не равен ontology-LOO}: в первом знакомые онтологии
      остаются в обучении; отдельная ontology-LOO-абляция даёт F1 $0{,}318$ без порога.
      Общие семь онтологий создают зависимость пар, не учтённую парными тестами

    \vspace{0.3em}
    \item \structure{Синтетические вариации}~--- та же онтология с переименованными
      идентификаторами; новых концептов они не добавляют
  \end{itemize}
\end{frame}

\begin{frame}{Таксономии или онтологии? Что является результатом?}
  \begin{itemize}
    \item \structure{Исходные данные~--- OWL-онтологии OAEI}, не «голые деревья».
      Рассматриваются классы и граф is-a; используются также доступные
      названия, атрибуты, комментарии и объявленная несовместимость.
    \item \structure{Вход matcher-а}: две иерархии в JSON, возможны несколько
      родителей. Корректный граф должен быть ациклическим.
    \item \structure{Выход}: частичное соответствие классов с оценками уверенности,
      не более одного партнёра у каждого класса; отдельно~--- отчёт о флагах.
    \item \structure{Результат сохраняется} в JSON-кэше \texttt{results/},
      поле \texttt{match\_pairs}. Эталон \texttt{alignments.json}
      нужен для оценки и калибровки, не для решения отложенной пары.
  \end{itemize}
  \begin{block}{Вклад в рамках НИР, не новая постановка задачи}
    Экспериментальный прототип, адаптация существующих методов, сравнение
    по общему протоколу, анализ постобработки и частичные структурные проверки.
    Сопоставления свойств и полного логического вывода нет.
  \end{block}
\end{frame}

% Приложение: сильный baseline и ложные соответствия.
\begin{frame}{Почему StringEquiv силён, а порог нужен?}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \structure{\large Почему совпадают названия}
      \begin{itemize}
        \item Один домен: организация научных конференций
        \item Эквивалентные классы часто имеют одинаковые метки
        \item StringEquiv: precision $0{,}883$, recall $0{,}551$,
          F1 $0{,}663$ (средние по 21 паре)
        \item Не распознаёт синонимы и перевод;
          общие аббревиатуры могут совпадать
      \end{itemize}
    \column{0.48\textwidth}
      \structure{\large Почему возникают ошибки}
      \begin{itemize}
        \item У \texttt{Reviewer} может не быть эквивалента среди
          \texttt{Paper} и \texttt{Conference}
        \item Жадный отбор сортирует все пары по сходству:
          принимается пара с двумя свободными узлами
        \item Конфликты могут оставить менее похожего партнёра;
          без порога получается слишком много предсказаний
        \item Порог допускает отказ. Высокое сходство
          не доказывает эквивалентность
      \end{itemize}
  \end{columns}
  \begin{block}{Почему не отказаться от LLM вообще?}
    На одних парах гибрид лучше строк, на других хуже.
    Отсутствие преимущества в этом эксперименте не означает
    ненужность LLM на других данных.
  \end{block}
\end{frame}

% Приложение: метрики и калибровка без эталона оцениваемой пары.
\begin{frame}{Как выбирается порог без подсказки из эталона?}
  \begin{enumerate}
    \item Откладываем одну пару онтологий из 21.
    \item Для каждого порога считаем средний F1 на остальных 20 парах.
    \item Выбираем лучший порог для данной модели и применяем к отложенной паре.
    \item Повторяем для всех пар; усредняем 21 полученный F1.
  \end{enumerate}
  \begin{block}{Leave-one-pair-out, не обучение BERT}
    Эталоны остальных пар участвуют в выборе порога.
    Эталон отложенной пары нужен только для оценки после выбора.
    Веса sentence-transformer-моделей не изменяются.
  \end{block}
  \[
    P=\frac{TP}{TP+FP},\qquad R=\frac{TP}{TP+FN},\qquad
    F_1=\frac{2PR}{P+R}.
  \]
  {\footnotesize Основные таблицы~--- средние метрик по парам (macro).
  Pooled-метрики считаются по суммарным TP/FP/FN и могут отличаться.
  Подбор лучшего порога на самой оцениваемой паре дал бы оптимистичную оценку.}
\end{frame}

% Приложение: статистическое превосходство и ограничения выборки.
\begin{frame}{Лучший средний F1 означает надёжно лучший метод?}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \structure{\large ruRoberta с порогом против StringEquiv}
      \begin{itemize}
        \item Средние: $0{,}677$ против $0{,}663$
        \item Средняя разница: $+0{,}014$
        \item Восемь побед, четыре ничьи, девять поражений
        \item Bootstrap-интервал разницы:
          $[-0{,}020;\,+0{,}051]$ (95\%), включает ноль
        \item Wilcoxon: $p=0{,}644$; $p_{\rm Holm}=1{,}00$
      \end{itemize}
    \column{0.48\textwidth}
      \structure{\large Как читать эти числа}
      \begin{itemize}
        \item Большие выигрыши на нескольких парах могут поднять среднее,
          не давая устойчивого превосходства
        \item Значимость превосходства не установлена;
          равенство методов тоже не доказано
        \item Семейство из 12 сравнений: поправка Холма
        \item 21 пара использует общие семь онтологий;
          эта зависимость в парных тестах не учтена
      \end{itemize}
  \end{columns}
  \begin{block}{Что подтвердилось в использованном протоколе}
    После поправки значимы эффект порога LaBSE относительно результата без порога
    и отставание BM25-LLM от строкового baseline в среднем по C/CP/CC.
    Перенос выводов на новый домен не проверен.
  \end{block}
\end{frame}

% Приложение: что действительно проверяет эксперимент с GNN.
\begin{frame}{Почему GraphSAGE не доказал пользу структуры?}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \structure{\large Наблюдение}
      \begin{tabular}{@{}lrr@{}}
        \toprule
        Метод & Без порога & LOO-порог \\
        \midrule
        MiniLM & 0{,}349 & 0{,}653 \\
        GraphSAGE & 0{,}333 & 0{,}667 \\
        \bottomrule
      \end{tabular}
      \par\vspace{0.6em}
      \begin{itemize}
        \item Исходные признаки~--- MiniLM, без его дообучения
        \item Обучали GraphSAGE и контрольную LinearSiamese
        \item Loss: косинус правильных пар $\to 1$,
          случайных отрицательных $\to 0$
        \item Парного теста разницы после калибровки нет
      \end{itemize}
    \column{0.48\textwidth}
      \structure{\large Ограничения реализации}
      \begin{itemize}
        \item Рёбра превращены в неориентированные:
          отдельные роли родителя и ребёнка не передаются
        \item Loss не оптимизирует непосредственно итоговый жадный отбор и F1
        \item Эквивалентные классы могут иметь разное окружение
        \item Мало эталонных соответствий: 258
      \end{itemize}
      \begin{block}{Причины не установлены}
        Эти ограничения могут объяснять результат, но причинного эксперимента нет.
        «Структура бесполезна» из этих чисел не следует.
      \end{block}
  \end{columns}
\end{frame}

% Приложение: потолок retrieval и потери последующих этапов.
\begin{frame}{Почему высокий recall@k не гарантирует итоговый recall?}
  \begin{block}{Условный пример: десять эталонных соответствий}
    Ретривер предложил девять правильных кандидатов.
    LLM отвергла три из них; ещё два потерялись при последующем отборе.
    Осталось четыре: $R_{\rm retrieval}=0{,}9$, $R_{\rm final}=0{,}4$.
  \end{block}
  \[
    \text{retrieval}\ \longrightarrow\ \text{LLM}\ \longrightarrow\
    \text{порог и отбор 1:1},\qquad R_{\rm final}\leq R_{\rm retrieval}.
  \]
  {\footnotesize Неравенство относится к одной и той же паре и набору кандидатов,
  если итоговые соответствия выбираются только из него.}
  \vspace{0.6em}
  \begin{itemize}
    \item Гибрид: recall@5 $0{,}927$, recall@10 $0{,}956$;
      MiniLM@5 $0{,}783$, BM25@5 $0{,}730$
    \item В основных LLM-прогонах: гибрид до десяти кандидатов, остальные до пяти
    \item Развёртка $k$ оценивает retrieval; качество итогового LLM-matcher-а
      при разных $k$ не измерено
    \item Разницу гибрида и BM25 нельзя целиком приписать объединению retriever-ов:
      бюджет кандидатов также различается
  \end{itemize}
\end{frame}

% Приложение: конкретные примеры структурных флагов.
\begin{frame}{Какой конфликт обнаруживает проверка согласованности?}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \structure{\large Объявленная несовместимость}
      \[
        A\perp B,\qquad A\leftrightarrow X,\quad B\leftrightarrow Y,
        \qquad Y\sqsubseteq X.
      \]
      \begin{itemize}
        \item Несовместимые исходные классы сопоставлены
          с классами, связанными подчинением
        \item Код помечает оба соответствия как
          \texttt{disjointness-violation} (error)
        \item Проверяются объявленные несовместимости обеих таксономий
      \end{itemize}
    \column{0.48\textwidth}
      \structure{\large Несохранение подчинения}
      \[
        \text{Статья}\sqsubseteq\text{Документ},
      \]
      \[
        \text{Статья}\leftrightarrow\texttt{Paper},\qquad
        \text{Документ}\leftrightarrow\texttt{Person}.
      \]
      \begin{itemize}
        \item Если Paper не потомок Person, ставится
          \texttt{subsumption-violation} (warning)
        \item Не обязательно инверсия: нужная связь может просто отсутствовать
        \item Это не доказывает ошибку соответствия при неполных данных
      \end{itemize}
  \end{columns}
  \begin{block}{Флаг~--- не остановка программы}
    Error/warning~--- уровни записи в отчёте.
    Проверка сама не удаляет соответствия; семантический фильтр исследован отдельно.
    Полный OWL-reasoner не используется.
  \end{block}
\end{frame}

% Приложение: избыточные проверки и пределы структурной диагностики.
\begin{frame}{Что гарантирует отсутствие флагов?}
  \begin{itemize}
    \item \structure{Не правильность сопоставления}: два класса можно перепутать,
      не создав ни одного из проверяемых структурных конфликтов.
    \item \structure{Не OWL-согласованность}: проверены граф и объявленная
      несовместимость, не все аксиомы и их логические следствия.
    \item \structure{Циклы возможны в JSON}: списки родителей допускают
      $A\to B\to A$; загрузчик не гарантирует DAG.
    \item \structure{Достижимость любого корня избыточна}, если граф конечен,
      ацикличен и все родители существуют. Корни определены как узлы без родителей;
      оторванная ветка может стать новым корнем и пройти проверку.
    \item \structure{Одинаковые названия не всегда ошибка}: это warning,
      контекстное различие классов не устанавливается.
    \item \structure{Инъективность результатов обеспечена отбором}.
      Её проверка полезна для произвольного Alignment, но ноль нарушений
      не является независимым доказательством качества.
  \end{itemize}
  {\footnotesize Путь к заранее заданному множеству допустимых корней был бы
  другим условием; в этой работе такого требования нет.}
\end{frame}

% Приложение: фактическая единица API-запроса и масштабирование.
\begin{frame}{Сколько запросов нужно и как масштабировать LLM?}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \structure{\large В нашей реализации}
      \begin{itemize}
        \item Один запрос на исходный узел со списком до $k$ кандидатов
        \item Для C/CP/CC~--- не более $3n$ первичных запросов;
          ансамбль объединяет ответы без нового вызова
        \item Retry увеличивает число попыток; пустой список кандидатов
          позволяет пропустить запрос
        \item $k$ увеличивает объём промпта и стоимость обработки,
          но не число первичных запросов на узел
      \end{itemize}
    \column{0.48\textwidth}
      \structure{\large Где остаётся стоимость}
      \begin{itemize}
        \item $n$ исходных, $m$ целевых узлов:
          полный перебор эмбеддинговых сходств~--- $O(nm)$
        \item Проверяется до $O(kn)$ пар-кандидатов на режим,
          сгруппированных в запросы
        \item Blocking или индекс для поиска соседей могут уменьшить поиск;
          это направление работы, не измеренное ускорение
        \item Большие таксономии экспериментально не проверены
      \end{itemize}
  \end{columns}
  \begin{block}{Ограничение надёжности API}
    При сбое предусмотрен резервный эмбеддинговый ответ с уверенностью $0{,}75$.
    Его происхождение не сохранено в конечном кэше: долю fallback восстановить нельзя.
    Не каждый ответ с $0{,}75$ является резервным.
  \end{block}
\end{frame}

% Приложение: честный ответ о незакрытых замечаниях.
\begin{frame}{Какие замечания и вопросы остаются открытыми?}
  \begin{itemize}
    \item \structure{Дообучение sentence-transformer-моделей не выполнено}.
      Обучение GraphSAGE не заменяет этот эксперимент.
    \item \structure{Итоговая чувствительность LLM к top-k не измерена}.
      Есть только развёртка recall ретривера.
    \item \structure{Нет экспериментального сравнения с AML, LogMap
      и классическими taxonomy-matcher-ами}.
      Новую постановку задачи или превосходство над областью не заявляем.
    \item \structure{Нет проверки переноса на новый предметный домен}.
      Ontology-LOO для GNN откладывает одну онтологию того же Conference-домена.
    \item \structure{Нет полного набора повторных LLM-прогонов},
      доказательства снижения дисперсии ансамблем и парного теста GNN после порога.
    \item \structure{Нет логического ремонта отображения}.
      Сочетание калиброванного порога с семантическим фильтром отдельно не оценено.
  \end{itemize}
  \begin{block}{Какой вывод защищается}
    Реализованы сопоставление и частичная структурная проверка.
    Измерены выбранные методы и влияние постобработки.
    Ограничения не выдаются за выполненные эксперименты.
  \end{block}
\end{frame}
